def decode_csv(payload: str):
    data = gzip.decompress(base64.b64decode(payload)).decode("utf-8-sig")
    return list(csv.DictReader(io.StringIO(data)))


def make_session():
    s = requests.Session()
    retry = Retry(total=5, connect=5, read=5, backoff_factor=.8,
                  status_forcelist=(429,500,502,503,504), allowed_methods=frozenset(["GET"]))
    s.mount("https://", HTTPAdapter(max_retries=retry))
    s.headers.update({
        "User-Agent":"Mozilla/5.0 FLB-17101-GitHub-Pages/1.0",
        "Accept":"image/avif,image/webp,image/apng,image/*,*/*;q=0.8",
    })
    return s


def valid(path: Path):
    try:
        with Image.open(path) as im:
            im.verify()
        return path.stat().st_size > 500
    except Exception:
        return False


def candidates(row):
    eid, rel = row["ElementID"], row["ImageURL"]
    return [
        f"https://www.lego.com{rel}",
        f"https://www.lego.com/cdn/product-assets/element.img.lod5photo.192x192/{eid}.jpg",
        f"https://images.brickset.com/parts/{eid}.jpg",
        f"https://images.brickset.com/parts/large/{eid}.jpg",
        f"https://cdn.rebrickable.com/media/parts/elements/{eid}.jpg",
    ]


def save_image(raw: bytes, path: Path):
    with Image.open(io.BytesIO(raw)) as im:
        im.load()
        im = im.convert("RGBA")
        im.thumbnail((640,640), Image.Resampling.LANCZOS)
        im.save(path, "WEBP", quality=92, method=6)


def download(row):
    path = IMAGES / f"{row['ElementID']}.webp"
    if valid(path):
        return {"element":row["ElementID"],"status":"cached","source":"repo"}
    s = make_session()
    errors = []
    for url in candidates(row):
        try:
            r = s.get(url, timeout=(15,45), allow_redirects=True)
            if r.status_code != 200:
                errors.append(f"{url} HTTP {r.status_code}")
                continue
            save_image(r.content, path)
            if valid(path):
                return {
                    "element":row["ElementID"],"status":"downloaded","source":r.url,
                    "sha256":hashlib.sha256(path.read_bytes()).hexdigest(),
                }
        except Exception as exc:
            errors.append(f"{url} {type(exc).__name__}: {exc}")
        time.sleep(.12)
    return {"element":row["ElementID"],"status":"failed","source":None,"errors":errors}

def rgb(value):
    value = value.lstrip("#")
    return tuple(int(value[i:i+2],16) for i in (0,2,4))


def light(value, amount=.78):
    return tuple(round(c+(255-c)*amount) for c in rgb(value))

def main():
    inventory, layout = decode_csv(INVENTORY_B64), decode_csv(LAYOUT_B64)
    by_key = {(r["DesignID"],r["Colour"]):r for r in inventory}
    parts=[]
    for h in layout:
        r=by_key[(h["design_id"],h["colour"])]
        parts.append({**h,**r})
    assert len(inventory)==218 and sum(int(r["Qty"]) for r in inventory)==850
    assert len(parts)==218

    results=[]
    with ThreadPoolExecutor(max_workers=8) as pool:
        futures={pool.submit(download,row):row for row in inventory}
        for i,fut in enumerate(as_completed(futures),1):
            result=fut.result(); results.append(result)
            print(f"[{i:03d}/218] {result['element']} {result['status']}",flush=True)

    success=[r for r in results if r["status"] in {"cached","downloaded"}]
    failed=[r for r in results if r["status"]=="failed"]
    summary={
        "total":len(results),"success":len(success),"failed":len(failed),
        "inventory_pieces":850,"design_colour_homes":218,
        "generated_at_utc":time.strftime("%Y-%m-%dT%H:%M:%SZ",time.gmtime()),
    }
    (REPORTS/"verification.json").write_text(json.dumps({"summary":summary,"results":sorted(results,key=lambda x:x["element"])},ensure_ascii=False,indent=2),encoding="utf-8")
    if failed:
        print(json.dumps(failed,ensure_ascii=False,indent=2))
        raise SystemExit(f"Image verification failed: {len(failed)} / 218")

    compose_proof(parts)
    tray_paths=[compose_tray(t,parts) for t in "ABC"]
    with Image.open(tray_paths[0]) as first:
        overview=Image.new("RGB",(first.width*3,first.height),(255,255,255))
    for i,p in enumerate(tray_paths):
        with Image.open(p) as im: overview.paste(im,(i*im.width,0))
    overview.save(STATIC/"overview.png",optimize=True)
    build_html(parts,summary)
    (ROOT/"README.md").write_text(
        "# LEGO BOOST 17101 self-hosted review\n\n"
        "Generated by GitHub Actions. All 218 part images are downloaded into `images/` and verified with Pillow.\n\n"
